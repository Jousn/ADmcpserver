HaltRequested: Boolean;

procedure TForm_PlaceSilk.Form_PlaceSilkKeyDown(Sender: TObject; var Key: Word; Shift: TShiftState);
begin
    if (ssCtrl in Shift) and ((Key = Ord('P')) or (Key = Ord('p'))) then
    begin
        HaltRequested := True;

        HintLbl.Visible := True;

// .pas 两个循环中添加
